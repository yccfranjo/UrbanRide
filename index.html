<script>
// --- 1. INITIALISATION MOTEUR 3D ---
const scene = new THREE.Scene();
scene.fog = new THREE.FogExp2(0x87CEEB, 0.015);
const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
camera.position.set(0, 5, 10);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(window.innerWidth, window.innerHeight);
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;
document.getElementById('canvas-container').appendChild(renderer.domElement);

const gltfLoader = new THREE.GLTFLoader();

// --- 2. LUMIÈRES & ENVIRONNEMENT ---
const hemiLight = new THREE.HemisphereLight(0xffffff, 0x444444, 0.6);
scene.add(hemiLight);
const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
dirLight.position.set(50, 100, 50);
dirLight.castShadow = true;
scene.add(dirLight);

// Sol du quartier (Asphalte)
const groundGeo = new THREE.PlaneGeometry(500, 500);
const groundMat = new THREE.MeshStandardMaterial({ color: 0x333333, roughness: 0.8 });
const ground = new THREE.Mesh(groundGeo, groundMat);
ground.rotation.x = -Math.PI / 2;
ground.receiveShadow = true;
scene.add(ground);

// Génération procédurale d'un quartier de base (Bâtiments)
const buildingMat = new THREE.MeshStandardMaterial({ color: 0x888888, roughness: 0.9 });
for(let i=0; i<30; i++) {
const h = Math.random() * 20 + 10;
const geo = new THREE.BoxGeometry(10, h, 10);
const building = new THREE.Mesh(geo, buildingMat);
building.position.set((Math.random() - 0.5) * 200, h/2, (Math.random() - 0.5) * 200);
if(building.position.length() > 20) { // Laisser le centre vide pour jouer
building.castShadow = true;
building.receiveShadow = true;
scene.add(building);
}
}

// --- 3. CHARGEMENT DES VRAIS VÉHICULES ---
const vehicles = [];
function spawnVehicle(filename) {
document.getElementById('loading').style.display = 'block';

// Le code cherche le fichier 3D dans le même dossier que ce site web
gltfLoader.load(filename, function(gltf) {
const car = gltf.scene;
car.position.set((Math.random() - 0.5) * 10, 0, (Math.random() - 0.5) * 10);
car.scale.set(1, 1, 1); // Ajuster l'échelle si le modèle est trop grand
car.traverse(function(child) { if (child.isMesh) child.castShadow = true; });
scene.add(car);
vehicles.push(car);
document.getElementById('loading').style.display = 'none';
}, undefined, function(error) {
// Si le vrai fichier n'est pas trouvé dans le dossier, on crée une boîte par défaut pour éviter le crash
console.error("Fichier 3D non trouvé : " + filename + ". Assure-toi de l'avoir uploadé.");
const fallbackGeo = new THREE.BoxGeometry(2, 1, 4);
const fallbackMat = new THREE.MeshStandardMaterial({ color: Math.random() * 0xffffff });
const fallbackCar = new THREE.Mesh(fallbackGeo, fallbackMat);
fallbackCar.position.set((Math.random() - 0.5) * 10, 0.5, (Math.random() - 0.5) * 10);
fallbackCar.castShadow = true;
scene.add(fallbackCar);
vehicles.push(fallbackCar);
document.getElementById('loading').innerHTML = "Erreur: Fichier 3D introuvable. Modèle de remplacement généré.";
setTimeout(() => document.getElementById('loading').style.display = 'none', 2000);
});
}

// --- 4. SYSTÈME DE FEU (PARTICULES RÉALISTES) ---
const fireParticles = [];
function spawnFire() {
const particleCount = 100;
const fireGeo = new THREE.BufferGeometry();
const firePos = new Float32Array(particleCount * 3);
for(let i=0; i<particleCount * 3; i++) firePos[i] = (Math.random() - 0.5) * 2;
fireGeo.setAttribute('position', new THREE.BufferAttribute(firePos, 3));

const fireMat = new THREE.PointsMaterial({
color: 0xff5500, size: 0.5, transparent: true, opacity: 0.8,
blending: THREE.AdditiveBlending, depthWrite: false
});

const fireSystem = new THREE.Points(fireGeo, fireMat);
// Apparaît devant la caméra
fireSystem.position.copy(camera.position);
fireSystem.position.z -= 5;
fireSystem.position.y = 0;
scene.add(fireSystem);
fireParticles.push(fireSystem);
}

// --- 5. SYSTÈME DE PNJ (NPC) ---
const npcs = [];
function spawnNPC() {
const npcGeo = new THREE.CylinderGeometry(0.5, 0.5, 2, 8);
const npcMat = new THREE.MeshStandardMaterial({ color: 0xffccaa });
const npc = new THREE.Mesh(npcGeo, npcMat);
npc.position.set(camera.position.x + (Math.random()*4 - 2), 1, camera.position.z - 5);
npc.castShadow = true;
scene.add(npc);
npcs.push(npc);
}

function callPolice() {
document.body.style.animation = "policeFlash 0.5s infinite";
setTimeout(() => document.body.style.animation = "", 3000);
alert("La BAC arrive, gaz lacrymogène déployé !");
scene.fog.color.setHex(0xaaaaaa); // Effet fumée blanche lacrymo
scene.fog.density = 0.1;
}

let isNight = false;
function toggleTime() {
isNight = !isNight;
if(isNight) {
scene.background = new THREE.Color(0x050510);
scene.fog.color.setHex(0x050510);
dirLight.intensity = 0.1;
hemiLight.intensity = 0.2;
} else {
scene.background = new THREE.Color(0x87CEEB);
scene.fog.color.setHex(0x87CEEB);
dirLight.intensity = 0.8;
hemiLight.intensity = 0.6;
scene.fog.density = 0.015;
}
}

// --- 6. BOUCLE D'ANIMATION ET PHYSIQUE ---
function animate() {
requestAnimationFrame(animate);

// Animation du feu
fireParticles.forEach(fire => {
const positions = fire.geometry.attributes.position.array;
for(let i=1; i<positions.length; i+=3) {
positions[i] += 0.05; // Les flammes montent
if(positions[i] > 3) positions[i] = 0;
}
fire.geometry.attributes.position.needsUpdate = true;
});

// Animation des PNJ (Marche aléatoire)
npcs.forEach(npc => {
npc.position.x += (Math.random() - 0.5) * 0.05;
npc.position.z += (Math.random() - 0.5) * 0.05;
});

renderer.render(scene, camera);
}
animate();

// Styles CSS injectés pour l'effet gyrophare
const style = document.createElement('style');
style.innerHTML = `@keyframes policeFlash { 0% { box-shadow: inset 0 0 50px blue; } 50% { box-shadow: inset 0 0 50px red; } 100% { box-shadow: inset 0 0 50px blue; } }`;
document.head.appendChild(style);

// Gestion de la redimension de l'écran
window.addEventListener('resize', () => {
camera.aspect = window.innerWidth / window.innerHeight;
camera.updateProjectionMatrix();
renderer.setSize(window.innerWidth, window.innerHeight);
});
</script>
